const { postToFacebook } = require('./facebook.cjs');
const { postToInstagram } = require('./instagram.cjs');
const { postToTikTok } = require('./tiktok.cjs');
const { postToYouTube } = require('./youtube.cjs');
const { postToWhatsAppChannel } = require('./whatsapp.cjs');

const fail = (m) => ({ ok: false, message: m });
const CONCURRENCY = 4;

async function postOne(p, text, file) {
  try {
    if (p === 'facebook') return await postToFacebook(text, file);
    if (p === 'instagram') return await postToInstagram(text, file);
    if (p === 'tiktok') return await postToTikTok(text, file);
    if (p === 'youtube') return await postToYouTube(text, file);
if (p === 'whatsapp') return await postToWhatsAppChannel(text, file);	

    if (p === 'whatsapp') {
      return fail(
        'WhatsApp Channel: automatic publishing is not configured yet. Open WhatsApp Web and post to your Channel manually.'
      );
    }

    return fail(p + ': posting is not supported.');
  } catch (e) {
    return fail(p + ': error - ' + e.message);
  }
}

async function postAll(text, file, platforms) {
  const results = new Array(platforms.length);
  let next = 0;

  async function worker() {
    while (next < platforms.length) {
      const i = next++;
      results[i] = await postOne(platforms[i], text, file);
    }
  }

  const n = Math.min(CONCURRENCY, platforms.length);
  await Promise.all(Array.from({ length: n }, worker));
  return results;
}

module.exports = { postAll };

async function attachFile(win, file) {
  dbgLog('---- attachFile shuru: ' + file);

  let mine = false;
  try { if (!dbg.isAttached()) { dbg.attach('1.3'); mine = true; } } catch (e) { dbgLog('debugger attach fail: ' + e.message); }
  let ok = false;

  const tryStep = async (name, fn) => {
    if (ok) return;
    try { await fn(); } catch (e) { dbgLog(name + ' error: ' + e.message); }
    await sleep(2500);
    ok = await hasPreview(win);
    dbgLog(name + ' -> ' + (ok ? 'PHOTO LAG GAYI' : 'nahi lagi'));
  };

  try {
    // 1) dialog ke andar wale file input mein
    await tryStep('1 file-input', async () => {
      await dbg.sendCommand('DOM.enable');
      const doc = await dbg.sendCommand('DOM.getDocument', { depth: -1, pierce: true });
      const q = await dbg.sendCommand('DOM.querySelectorAll', { nodeId: doc.root.nodeId, selector: 'div[role="dialog"] input[type="file"]' });
      const ids = (q.nodeIds || []).filter(Boolean);
      dbgLog('dialog ke andar file inputs: ' + ids.length);
      if (ids.length) await dbg.sendCommand('DOM.setFileInputFiles', { files: [file], nodeId: ids[ids.length - 1] });
    });

    // 2) "Add photos or videos" dabao aur file chooser intercept karo
    await tryStep('2 file-chooser', async () => {
      let handler = null;
      try {
        await dbg.sendCommand('Page.enable');
        await dbg.sendCommand('Page.setInterceptFileChooserDialog', { enabled: true });
        let handled = false;
        handler = async (ev, method, params) => {
          if (method === 'Page.fileChooserOpened' && !handled) {
            handled = true;
            dbgLog('file chooser khula, photo daal raha hoon');
            try { await dbg.sendCommand('DOM.setFileInputFiles', { files: [file], backendNodeId: params.backendNodeId }); } catch (e) { dbgLog('chooser set error: ' + e.message); }
          }
        };
        dbg.on('message', handler);
        const pos = await dropzonePos(win);
        if (!pos) { dbgLog('Add photos wala hissa nahi mila'); return; }
        await realClick(win, pos.x, pos.y);
        for (let i = 0; i < 6 && !handled; i++) await sleep(500);
        if (!handled) dbgLog('click ke baad file chooser nahi khula');
      } finally {
        try { if (handler) dbg.removeListener('message', handler); } catch (e) {}
        try { await dbg.sendCommand('Page.setInterceptFileChooserDialog', { enabled: false }); } catch (e) {}
      }
    });

    // 3) drag and drop
    await tryStep('3 drag-drop', async () => {
      const pos = await dropzonePos(win);
      if (!pos) { dbgLog('dropzone nahi mila'); return; }
      const data = { items: [], files: [file], dragOperationsMask: 1 };
      await dbg.sendCommand('Input.dispatchDragEvent', { type: 'dragEnter', x: pos.x, y: pos.y, data });
      await sleep(150);
      await dbg.sendCommand('Input.dispatchDragEvent', { type: 'dragOver', x: pos.x, y: pos.y, data });
      await sleep(150);
      await dbg.sendCommand('Input.dispatchDragEvent', { type: 'drop', x: pos.x, y: pos.y, data });
    });

    // 4) clipboard se paste (sirf photo ke liye)
    await tryStep('4 paste', async () => {
      const img = nativeImage.createFromPath(file);
      if (img.isEmpty()) { dbgLog('paste: file photo nahi hai (video?)'); return; }
      clipboard.writeImage(img);
      await js(win, `(function () {
        const t = document.querySelector('div[role="dialog"] div[contenteditable="true"][role="textbox"]') || document.querySelector('div[contenteditable="true"][role="textbox"]');
        if (t) t.focus();
        return !!t;
      })();`);

  } finally {

  }

function stripLinks(t) {
  const has = /https?:\/\/\S+|www\.\S+/i;
  const all = /https?:\/\/\S+|www\.\S+/gi;
  return String(t || '').split('\n').map((line) => {
    if (!has.test(line)) return line;
    const rest = line.replace(all, '').trim();
    return rest === '' || /^[^:\uFF1A]{0,30}[:\uFF1A]$/.test(rest) ? null : rest;
  }).filter((l) => l !== null).join('\n').replace(/\n{3,}/g, '\n\n').trim();
}

const MEDIA_OK = /\.(jpe?g|png|gif|webp|bmp|mp4|mov|m4v|webm|avi|mkv)$/i;
function mediaPathFromUrl(u) {
  const p = decodeURIComponent(new URL(u).pathname.slice(1));
  return MEDIA_OK.test(p) ? p : null;
}
function registerMedia() {
  if (typeof protocol.handle === 'function') {
    protocol.handle('media', (req) => {
      const p = mediaPathFromUrl(req.url);
      if (!p) return new Response('Forbidden', { status: 403 });
      return net.fetch(pathToFileURL(p).toString());
    });
  } else {
    protocol.registerFileProtocol('media', (req, cb) => {
      const p = mediaPathFromUrl(req.url);
      cb(p ? { path: p } : { error: -10 });
    });
  }
}

    const it = data.items[i];
    mainWin.webContents.send('batch-progress', { index: i, status: 'uploading' });
    let results;
    try {
      if (data.dryRun) {
        results = (it.platforms || data.platforms).map((p) => ({ ok: true, message: p + ': TEST MODE - asli post nahi hui' }));
      } else {
        const caption = [it.text, data.description].filter(Boolean).join('\n\n');
        results = await postAll(caption, it.file, (it.platforms || data.platforms));
      }
    } catch (err) {
      results = [{ ok: false, message: 'error - ' + err.message }];
    }
    const ok = results.every((r) => r.ok);
    mainWin.webContents.send('batch-progress', { index: i, status: ok ? 'done' : 'failed', results });
    out.push({ index: i, file: it.file, ok, results });
    await wait(3000);
  }
  return { items: out };
});

import { useState, useEffect, useRef } from 'react';
import PreviewCarousel from './PreviewCarousel'

const NAMES = ['facebook', 'instagram', 'tiktok', 'youtube', 'whatsapp'];
const isVid = (p) => /\.(mp4|mov|webm|mkv|avi|m4v)$/i.test(p || '');
const LBL = { waiting: 'Waiting', uploading: 'Uploading...', done: 'Done', failed: 'Failed' };

export default function MultiPost({ Dot, Media, addHistory }) {
  const [items, setItems] = useState([]);
  const [description, setDescription] = useState('');
  const [purpose, setPurpose] = useState('');
  const [platforms, setPlatforms] = useState({
    facebook: false, instagram: false, tiktok: false, youtube: false, whatsapp: false,
  });
  const [testMode, setTestMode] = useState(true);
  const [running, setRunning] = useState(false);
  const idxRef = useRef([]);

  useEffect(() => {
    window.api.onBatchProgress((m) => {
      const i = idxRef.current[m.index];
      if (i === undefined) return;
      const st = m.status === 'done' && !(m.results && m.results.length) ? 'failed' : m.status;
      setItems((prev) => prev.map((it, k) => (k === i ? { ...it, status: st, results: m.results || it.results } : it)));
    });
  }, []);

  const selected = NAMES.filter((p) => platforms[p]);
  const canRun = !running && items.length > 0 && selected.length > 0 && description.trim().length > 0
    && (testMode || items.some((it) => it.status !== 'done'));
  const hasFailed = items.some((it) => it.status === 'failed');

  const pick = async () => {
    const fs = await window.api.pickFiles();
    if (fs && fs.length) {
      setItems((prev) => [
        ...prev,
        ...fs.filter((f) => !prev.some((x) => x.file === f)).map((f) => ({ file: f, text: '', status: 'waiting' })),
      ]);
    }
  };

  const run = async (onlyFailed) => {
    const idx = items.map((_, i) => i).filter((i) => (onlyFailed ? items[i].status === 'failed' : (testMode || items[i].status !== 'done')));
    if (!idx.length) return;
    if (!testMode && !window.confirm('Live post hone wali hai!\n\n' + idx.length + ' item(s) -> ' + selected.join(', ') + '\n\nPublish karein?')) return;
    const payload = idx.map((i) => ({
      file: items[i].file,
      text: items[i].text,
      platforms: selected.filter((p) => !(p === 'youtube' && !isVid(items[i].file))),
    }));
    idxRef.current = idx;
    setItems((prev) => prev.map((it, i) => (idx.includes(i) ? { ...it, status: 'waiting', results: null } : it)));
    setRunning(true);
    let res = null;
    try {
      res = await window.api.postBatch({ items: payload, description, platforms: selected, dryRun: testMode });
    } catch (e) {
      res = null;
    }
    if (res && res.items) {
      const entries = [];
      res.items.forEach((o) => {
        const i = idx[o.index];
        const it = items[i];
        const pl = payload[o.index].platforms;
        (o.results || []).forEach((r, k) => entries.push({
          id: 'b' + Date.now() + '-' + i + '-' + k,
          platform: pl[k] || pl[0] || 'facebook',
          postType: isVid(it.file) ? 'reel' : 'post',
          purpose: purpose.trim(), text: it.text, description, file: it.file,
          ok: !!r.ok, message: r.message,
          postedAt: new Date().toISOString(), scheduledFor: null, dryRun: testMode,
        }));
      });
      if (entries.length) addHistory(entries);
    } else {
      setItems((prev) => prev.map((it, i) => (idx.includes(i) ? { ...it, status: 'failed' } : it)));
    }
    setRunning(false);
  };

  return (
    <div style={{ maxWidth: 900 }}>
      <div className="card">
        <h2>Multiple Post</h2>
        <p className="sub">Ek sath 6-7 photos/videos choose karein. Har ek alag post banegi, ek ek karke.</p>
        <div className="drop" onClick={pick}>
          <div className="thumb" />
          <div><b>Files choose karein</b><small>{items.length} selected (jpg, png, mp4, mov)</small></div>
        </div>
        {items.map((it, i) => (
          <div key={it.file} className={'item ' + (it.status === 'failed' ? 'bad' : '')}>
            <Media path={it.file} />
            <div className="grow">
              <div className="who">
                <b>{i + 1}. {it.file.split('\\').pop()}</b>
                <span className={'kindtag ' + (it.status === 'done' ? 'okt' : it.status === 'failed' ? 'badt' : '')}>{LBL[it.status]}</span>
              </div>
              <input
                type="text" className="in" placeholder="Is item ka caption (optional)" value={it.text}
                disabled={running}
                onChange={(e) => { const v = e.target.value; setItems((prev) => prev.map((x, k) => (k === i ? { ...x, text: v } : x))); }}
              />
              {it.status === 'failed' && it.results && (
                <small className="err">{it.results.filter((r) => !r.ok).map((r) => r.message).join(' | ')}</small>
              )}
            </div>
            <div className="acts">
              <button className="mini del" disabled={running} onClick={() => setItems((prev) => prev.filter((_, k) => k !== i))}>Remove</button>
            </div>
          </div>
        ))}
      </div>

      <div className="card">
        <h2>Description <em>*</em></h2>
        <p className="sub">Sab items ke liye same, required.</p>
        <textarea className="in sm" placeholder="Description..." value={description} onChange={(e) => setDescription(e.target.value)} />
        <div className="stack" style={{ marginTop: 12 }}>
          <input type="text" className="in" placeholder="Purpose (note, history mein dikhega)" value={purpose} onChange={(e) => setPurpose(e.target.value)} />
        </div>
      </div>

      <div className="card">
        <h2>Where would you like to post?</h2>
        <div className="plats">
          {NAMES.map((p) => (
            <div key={p} className={'plat' + (platforms[p] ? ' on' : '')} onClick={() => setPlatforms({ ...platforms, [p]: !platforms[p] })}>
              <Dot p={p} />{p}
            </div>
          ))}
        </div>
        {platforms.youtube && items.some((it) => !isVid(it.file)) && (
          <div className="warn">YouTube sirf video leta hai, photos YouTube pe skip hongi.</div>
        )}
        <label className="tg">
          <input type="checkbox" checked={testMode} onChange={(e) => setTestMode(e.target.checked)} />
          <span className="sw" />
          <span><b>Test mode</b><small>{testMode ? 'ON: asli post nahi hogi' : 'OFF: live post hogi!'}</small></span>
        </label>
        <button className="go" disabled={!canRun} onClick={() => run(false)}>
          {running ? 'Posting... (window band na karein)' : testMode ? 'Run Test (' + items.length + ' items)' : 'Publish All (' + items.length + ' items)'}
        </button>
        {hasFailed && !running && (
          <button className="mini" style={{ marginTop: 10 }} onClick={() => run(true)}>Sirf failed wale dobara try karein</button>
        )}
      </div>
    </div>
  );
}

const { makeWindow, sleep } = require('./util.cjs');

const WHATSAPP_CHANNEL_URL =
  'https://web.whatsapp.com/accept?code=0029VbDorTg2f3ED1P0rjy2H';

async function postToWhatsApp(caption, filePath) {
  const win = makeWindow('whatsapp');

  try {
    // WhatsApp Web open karein
    await win.loadURL('https://web.whatsapp.com');

    // Login aur page load hone ka waqt dein
    await sleep(5000);

    // Apne Channel ka invite link open karein
    await win.loadURL(WHATSAPP_CHANNEL_URL);

    // Channel page load hone ka waqt dein
    await sleep(5000);

    return {
      ok: false,
      message:
        'WhatsApp Channel opened. Automatic posting is not implemented yet.',
      requiresManualPost: true
    };
  } catch (error) {
    return {
      ok: false,
      message: `WhatsApp error: ${error.message}`
    };
  }
}

async function postAll(caption, filePath, platforms) {
  const results = [];

  if (platforms.includes('whatsapp')) {
    const result = await postToWhatsApp(caption, filePath);

    results.push({
      platform: 'whatsapp',
      ...result
    });
  }

  return results;
}
