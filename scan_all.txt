===== electron\bots.cjs =====
const { postToFacebook } = require('./facebook.cjs');
const { postToInstagram } = require('./instagram.cjs');
const { postToTikTok } = require('./tiktok.cjs');
const { postToYouTube } = require('./youtube.cjs');

const fail = (m) => ({ ok: false, message: m });

async function postAll(text, file, platforms) {
  const results = [];
  for (const p of platforms) {
    try {
      if (p === 'facebook') results.push(await postToFacebook(text, file));
      else if (p === 'instagram') results.push(await postToInstagram(text, file));
      else if (p === 'tiktok') results.push(await postToTikTok(text, file));
      else if (p === 'youtube') results.push(await postToYouTube(text, file));
      else results.push(fail(p + ': abhi post support nahi (sirf login hota hai)'));
    } catch (e) {
      results.push(fail(p + ': error - ' + e.message));
    }
  }
  return results;
}

module.exports = { postAll };

===== electron\facebook.cjs =====
const { BrowserWindow } = require('electron');
const fs = require('fs');

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const DIR = 'C:\\SocialPosterData';

async function shot(win, name) {
  try {
    fs.mkdirSync(DIR, { recursive: true });
    const img = await win.webContents.capturePage();
    fs.writeFileSync(DIR + '\\' + name + '.png', img.toPNG());
  } catch (e) {}
}

async function attachFile(win, file) {
  const pos = await js(win, `(function(){const b=document.querySelector("div[role=dialog] div[aria-label*=Photo]");if(!b)return null;const r=b.getBoundingClientRect();return {x:Math.round(r.left+r.width/2),y:Math.round(r.top+r.height/2)};})();`);
  if (pos) { await realClick(win, pos.x, pos.y); await sleep(2500); }
  const dbg = win.webContents.debugger;
  try { dbg.attach("1.3"); } catch (e) {}
  try {
    const doc = await dbg.sendCommand("DOM.getDocument", { depth: -1 });
    let q = await dbg.sendCommand("DOM.querySelector", { nodeId: doc.root.nodeId, selector: `input[type="file"]` });
    if (!q.nodeId) q = await dbg.sendCommand("DOM.querySelector", { nodeId: doc.root.nodeId, selector: `input[type="file"]` });
    if (q.nodeId) await dbg.sendCommand("DOM.setFileInputFiles", { files: [file], nodeId: q.nodeId });
  } catch (e) {}
  try { dbg.detach(); } catch (e) {}
  await sleep(8000);
  await shot(win, "fb-3b-attached");
}
function js(win, code) {
  return win.webContents.executeJavaScript(code);
}

async function realClick(win, x, y) {
  const wc = win.webContents;
  wc.sendInputEvent({ type: 'mouseMove', x, y });
  await sleep(200);
  wc.sendInputEvent({ type: 'mouseDown', x, y, button: 'left', clickCount: 1 });
  wc.sendInputEvent({ type: 'mouseUp', x, y, button: 'left', clickCount: 1 });
}

async function postToFacebook(text, file) {
  const win = new BrowserWindow({
    width: 1400, height: 950, show: true,
    webPreferences: { partition: 'persist:facebook' },
  });
  win.webContents.setUserAgent(
    'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/' + process.versions.chrome.split('.')[0] + '.0.0.0 Safari/537.36'
  );
  try {
    await win.loadURL('https://www.facebook.com/');
    await sleep(8000);

    // box ki asli position (sab se chhota element jis mein "on your mind" likha ho)
    const pos = await js(win, `
      (function () {
        const all = [...document.querySelectorAll('div[role="button"], span')];
        const m = all.filter(e => /on your mind/i.test((e.innerText || '').trim()) && (e.innerText || '').length < 100);
        if (!m.length) return null;
        const el = m[m.length - 1];
        el.scrollIntoView({ block: 'center' });
        const r = el.getBoundingClientRect();
        return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) };
      })();
    `);
    await sleep(800);
    await shot(win, 'fb-1-home');
    if (!pos) return { ok: false, message: 'Facebook: post box nahi mila (fb-1-home.png dekhein)' };

    const pos2 = await js(win, `
      (function () {
        const all = [...document.querySelectorAll('div[role="button"], span')];
        const m = all.filter(e => /on your mind/i.test((e.innerText || '').trim()) && (e.innerText || '').length < 100);
        const el = m[m.length - 1];
        const r = el.getBoundingClientRect();
        return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) };
      })();
    `);
    await realClick(win, pos2.x, pos2.y);

    // dialog ka intezar
    let dlg = false;
    for (let i = 0; i < 12; i++) {
      await sleep(1000);
      dlg = await js(win, `!!document.querySelector('div[contenteditable="true"][role="textbox"]')`).catch(() => false);
      if (dlg) break;
    }
    await shot(win, 'fb-2-dialog');
    if (!dlg) return { ok: false, message: 'Facebook: click ke baad bhi dialog nahi khuli (fb-2-dialog.png dekhein)' };

    await js(win, `document.querySelector('div[contenteditable="true"][role="textbox"]').focus(); true;`);
    await sleep(500);
    await win.webContents.insertText(text || '');
    if (file) await attachFile(win, file);
    await sleep(3000);
    await shot(win, 'fb-3-typed');

    const btnPos = await js(win, `
      (function () {
        const b = document.querySelector('div[aria-label="Post"][role="button"]');
        if (!b) return 'nobutton';
        if (b.getAttribute('aria-disabled') === 'true') return 'disabled';
        const r = b.getBoundingClientRect();
        return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) };
      })();
    `);
    if (typeof btnPos === 'string') return { ok: false, message: 'Facebook: Post button ' + btnPos + ' (fb-3-typed.png dekhein)' };
    await realClick(win, btnPos.x, btnPos.y);
    await sleep(3000);
    // "Add WhatsApp button" jaisa popup aaye to "Not now" dabao, phir Post dobara
    const notNow = await js(win, `
      (function () {
        const b = [...document.querySelectorAll('div[role="button"], button')]
          .find(e => /^not now$/i.test((e.innerText || '').trim()));
        if (!b) return null;
        const r = b.getBoundingClientRect();
        return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) };
      })();
    `);
    if (notNow) {
      await realClick(win, notNow.x, notNow.y);
      await sleep(2000);
      const again = await js(win, `
        (function () {
          const b = document.querySelector('div[aria-label="Post"][role="button"]');
          if (!b || b.getAttribute('aria-disabled') === 'true') return null;
          const r = b.getBoundingClientRect();
          return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) };
        })();
      `);
      if (again) await realClick(win, again.x, again.y);
    }
    await sleep(8000);
    await shot(win, 'fb-4-after');

    const still = await js(win, `!!document.querySelector('div[contenteditable="true"][role="textbox"]')`).catch(() => true);
    if (still) return { ok: false, message: 'Facebook: Post dabaya lekin dialog band nahi hui (fb-4-after.png dekhein)' };
    return { ok: true, message: 'Facebook: dialog band hui, profile pe check karein' };
  } finally {
    await sleep(5000);
    win.close();
  }
}

module.exports = { postToFacebook };

  wc.sendInputEvent({ type: 'mouseMove', x, y });
  await sleep(200);
  wc.sendInputEvent({ type: 'mouseDown', x, y, button: 'left', clickCount: 1 });
  wc.sendInputEvent({ type: 'mouseUp', x, y, button: 'left', clickCount: 1 });
}

const { sleep, makeWindow, waitFor, clickText, setFile, ensureLogin } = require('./util.cjs');

const { execFileSync } = require('child_process');
function imageToVideo(img) {
  const ff = require('ffmpeg-static');
  const out = 'C:\\SocialPosterData\\tiktok-photo.mp4';
  require('fs').mkdirSync('C:\\SocialPosterData', { recursive: true });
  execFileSync(ff, ['-y','-loop','1','-i',img,'-f','lavfi','-i','anullsrc=r=44100:cl=stereo','-t','6','-vf','scale=1080:1920:force_original_aspect_ratio=decrease,pad=1080:1920:(ow-iw)/2:(oh-ih)/2:black,format=yuv420p','-r','30','-c:v','libx264','-c:a','aac','-shortest',out]);
  return out;
}
async function postToTikTok(text, file) {
  if (file && /\.(png|jpe?g|webp)$/i.test(file)) file = imageToVideo(file); if (!file || !/\.(mp4|mov|webm)$/i.test(file))
    return { ok: false, message: 'TikTok: sirf video (mp4/mov) choose karein' };
  const win = makeWindow('tiktok');
  try {
    await ensureLogin(win, 'https://www.tiktok.com/tiktokstudio/upload');

    if (!(await waitFor(win, `!!document.querySelector('input[type="file"]')`, 30000)))
      return { ok: false, message: 'TikTok: upload box nahi mila' };
    if (!(await setFile(win, 'input[type="file"]', file)))
      return { ok: false, message: 'TikTok: file attach nahi hui' };

    const captionOk = await waitFor(win, `
      (function () {
        const box = document.querySelector('div[contenteditable="true"]');
        if (!box) return false;
        box.focus();
        document.execCommand('selectAll');
        document.execCommand('insertText', false, ${JSON.stringify(text || '')});
        return true;
      })();
    `, 90000);
    if (!captionOk) return { ok: false, message: 'TikTok: caption box nahi mila' };

    // Video upload hone tak Post button disabled rehta hai, enabled hote hi click
    const clicked = await waitFor(win, `
      (function () {
        const b = [...document.querySelectorAll('button')].find(e =>
          /^post$/i.test((e.innerText || '').trim()) &&
          !e.disabled && e.getAttribute('aria-disabled') !== 'true' && e.getAttribute('data-disabled') !== 'true');
        if (!b) return false;
        b.click();
        return true;
      })();
    `, 240000);
    if (!clicked) return { ok: false, message: 'TikTok: Post button enable nahi hua' };

    await sleep(3000);
    await clickText(win, '^post now$'); // "Continue to post?" popup
    const done = await waitFor(win, `/uploaded|manage your posts|video is being processed/i.test(document.body.innerText)`, 60000);
    return done
      ? { ok: true, message: 'TikTok: post ho gayi' }
      : { ok: false, message: 'TikTok: Post dabaya lekin confirm nahi aaya, check karein' };
  } finally {
    await sleep(2000);
    win.close();
  }
}

module.exports = { postToTikTok };

===== electron\util.cjs =====
const { BrowserWindow } = require('electron');

const UA = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/' + process.versions.chrome.split('.')[0] + '.0.0.0 Safari/537.36';

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

function makeWindow(name) {
  const win = new BrowserWindow({
    width: 1100,
    height: 780,
    show: true,
    webPreferences: { partition: 'persist:' + name },
  });
  win.webContents.setUserAgent(UA);
  return win;
}

async function waitFor(win, js, ms = 30000) {
  const end = Date.now() + ms;
  while (Date.now() < end) {
    try {
      if (await win.webContents.executeJavaScript(js)) return true;
    } catch (e) {}
    await sleep(1000);
  }
  return false;
}

function clickText(win, pattern, sel = 'div[role="button"], button, a') {
  return win.webContents.executeJavaScript(`
    (function () {
      const re = new RegExp(${JSON.stringify(pattern)}, 'i');
      const el = [...document.querySelectorAll(${JSON.stringify(sel)})]
        .find(e => re.test((e.innerText || '').trim()));
      if (!el) return false;
      el.click();
      return true;
    })();
  `);
}

async function setFile(win, selector, filePath) {
  const dbg = win.webContents.debugger;
  if (!dbg.isAttached()) dbg.attach('1.3');
  const { root } = await dbg.sendCommand('DOM.getDocument', { depth: -1 });
  const { nodeId } = await dbg.sendCommand('DOM.querySelector', {
    nodeId: root.nodeId,
    selector,
  });
  if (!nodeId) return false;
  await dbg.sendCommand('DOM.setFileInputFiles', { nodeId, files: [filePath] });
  return true;
}

// Login page ho to user ka 3 minute tak intezar (window band nahi hoti)
async function ensureLogin(win, url) {
  await win.loadURL(url);
  await sleep(6000);
  let waited = false;
  for (let i = 0; i < 90; i++) {
    const needs = await win.webContents
      .executeJavaScript(`!!document.querySelector('input[type="password"]') || /\\/login/i.test(location.href)`)
      .catch(() => true);
    if (!needs) break;
    waited = true;
    await sleep(2000);
  }
  if (waited) {
    await win.loadURL(url);
    await sleep(6000);
  }
}

module.exports = { UA, sleep, makeWindow, waitFor, clickText, setFile, ensureLogin };

const { sleep, makeWindow, waitFor, clickText, setFile, ensureLogin } = require('./util.cjs');

const run = (win, code) => win.webContents.executeJavaScript(code).catch(() => false);
const clickSel = (win, sel) =>
  run(win, `(function(){const e=document.querySelector(${JSON.stringify(sel)}); if(!e) return false; e.click(); return true;})()`);

async function postToYouTube(text, file) {
  if (!file || !/\.(mp4|mov|webm)$/i.test(file))
    return { ok: false, message: 'YouTube: sirf video choose karein' };
  const win = makeWindow('youtube');
  try {
    await ensureLogin(win, 'https://studio.youtube.com');
    await sleep(3000);

    if (!(await waitFor(win, `!!document.querySelector('#create-icon')`, 30000)))
      return { ok: false, message: 'YouTube: Create button nahi mila (login check karein)' };
    await clickSel(win, '#create-icon');
    await sleep(1500);
    await clickSel(win, '#text-item-0');

    if (!(await waitFor(win, `!!document.querySelector('input[type="file"]')`, 30000)))
      return { ok: false, message: 'YouTube: upload box nahi mila' };
    if (!(await setFile(win, 'input[type="file"]', file)))
      return { ok: false, message: 'YouTube: file attach nahi hui' };

    const title = (text || 'Video').split('\n')[0].slice(0, 95);
    const titleOk = await waitFor(win, `
      (function () {
        const box = document.querySelector('#title-textarea #textbox');
        if (!box) return false;
        box.focus();
        document.execCommand('selectAll');
        document.execCommand('insertText', false, ${JSON.stringify(title)});
        return true;
      })();
    `, 90000);
    if (!titleOk) return { ok: false, message: 'YouTube: title box nahi mila' };

    if (text) {
      await run(win, `
        (function () {
          const d = document.querySelector('#description-textarea #textbox');
          if (!d) return false;
          d.focus();
          document.execCommand('insertText', false, ${JSON.stringify(text)});
          return true;
        })();
      `);
    }

    await clickSel(win, '[name="VIDEO_MADE_FOR_KIDS_NOT_MFK"]');
    for (let i = 0; i < 3; i++) {
      await sleep(1500);
      await clickSel(win, '#next-button');
    }
    await sleep(1500);
    await clickSel(win, '[name="PUBLIC"]');
    await sleep(1500);
    const done = await clickSel(win, '#done-button');
    await sleep(10000);
    return done
      ? { ok: true, message: 'YouTube: upload ho gayi (Studio mein check kar lein)' }
      : { ok: false, message: 'YouTube: Done button nahi mila, check karein' };
  } finally {
    await sleep(2000);
    win.close();
  }
}

module.exports = { postToYouTube };
